    function shuffle(arr) {
      const a = [...arr];
      for (let i = a.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [a[i], a[j]] = [a[j], a[i]];
      }
      return a;
    }

      toastTimeout = setTimeout(() => el.classList.remove('show'), duration);
    }

    let memFlipped = [];
    let memLocked = false;
    let memScore = 0;
    let memCombo = 0;
    let memMatched = 0;
    let memStartTime = 0;
    const MEM_PAIRS = 6;

    function startMemoryGame(genre) {
      currentSub = genre;
      memFlipped = [];
      memLocked = false;
      memScore = 0;
      memCombo = 0;
      memMatched = 0;

      const labels = { geo: '地理', history: '歴史', civics: '公民', all: '全分野ミックス' };
      document.getElementById('mem-genre').textContent = labels[genre];
      document.getElementById('mem-score').textContent = '0';

      let pool = [];
      if (genre === 'all') {
        pool = [
          ...DB.memory.geo[currentDiff],
          ...DB.memory.history[currentDiff],
          ...DB.memory.civics[currentDiff]
        ];
      } else {
        pool = [...DB.memory[genre][currentDiff]];
      }

      const selected = shuffle(pool).slice(0, MEM_PAIRS);
      let cards = [];
      selected.forEach((item, idx) => {
        cards.push({ id: idx, type: 'a', text: item.a });
        cards.push({ id: idx, type: 'b', text: item.b });
      });
      cards = shuffle(cards);

      const grid = document.getElementById('memory-grid');
      grid.innerHTML = '';
      cards.forEach(c => {
        const el = document.createElement('div');
        el.className = 'm-card';
        el.innerHTML = `
          <div class="m-face m-back">？</div>
          <div class="m-face m-front type-${c.type}">${c.text}</div>
        `;
        el.onclick = () => tapMemoryCard(el, c);
        grid.appendChild(el);
      });

      showScreen('screen-memory');
      memStartTime = performance.now();
      timerInterval = setInterval(() => {
        const sec = ((performance.now() - memStartTime) / 1000).toFixed(1);
        document.getElementById('mem-time').textContent = sec;
      }, 100);
    }

    function tapMemoryCard(el, data) {
      if (memLocked || el.classList.contains('flipped') || el.classList.contains('matched')) return;
      el.classList.add('flipped');
      memFlipped.push({ el, data });

      if (memFlipped.length === 2) {
        memLocked = true;
        const [c1, c2] = memFlipped;
        if (c1.data.id === c2.data.id && c1.data.type !== c2.data.type) {
          memCombo++;
          const pts = 100 + (memCombo - 1) * 30;
          memScore += pts;
          memMatched++;
          document.getElementById('mem-score').textContent = memScore;
          showToast(`✨ 正解！ +${pts}点` + (memCombo > 1 ? ` (${memCombo}連続!)` : ''));

          setTimeout(() => {
            c1.el.classList.add('matched');
            c2.el.classList.add('matched');
            memFlipped = [];
            memLocked = false;
            if (memMatched === MEM_PAIRS) {
              clearInterval(timerInterval);
              const elapsed = (performance.now() - memStartTime) / 1000;
              const timeBonus = Math.max(0, Math.round((60 - elapsed) * 5));
              const total = memScore + timeBonus;
              finishGame('神経衰弱アタック', total, `クリアタイム: ${elapsed.toFixed(1)}秒（スピードボーナス +${timeBonus}点）`);
            }
          }, 550);
        } else {
          memCombo = 0;
          setTimeout(() => {
            c1.el.classList.remove('flipped');
            c2.el.classList.remove('flipped');
            memFlipped = [];
            memLocked = false;
          }, 950);
        }
      }
    }

    let kanjiQuestions = [];
    let kanjiIndex = 0;
    let kanjiScore = 0;
    let kanjiTimeLeft = 60;
    let kanjiInput = []; // { char, tileIdx }

    function startKanjiGame() {
      kanjiQuestions = shuffle(DB.kanji[currentDiff]).slice(0, 8);
      kanjiIndex = 0;
      kanjiScore = 0;
      kanjiTimeLeft = currentDiff === 'normal' ? 60 : 75;

      document.getElementById('kanji-score').textContent = '0';
      document.getElementById('kanji-timer').textContent = kanjiTimeLeft;

      showScreen('screen-kanji');
      renderKanjiQuestion();

      timerInterval = setInterval(() => {
        kanjiTimeLeft--;
        document.getElementById('kanji-timer').textContent = kanjiTimeLeft;
        if (kanjiTimeLeft <= 0) {
          clearInterval(timerInterval);
          finishGame('漢字組み立てクイズ', kanjiScore, 'タイムアップ！よくがんばりました！');
        }
      }, 1000);
    }

    function renderKanjiQuestion() {
      if (kanjiIndex >= kanjiQuestions.length) {
        clearInterval(timerInterval);
        const timeBonus = kanjiTimeLeft * 10;
        finishGame('漢字組み立てクイズ', kanjiScore + timeBonus, `全問クリア！残り時間ボーナス +${timeBonus}点`);
        return;
      }

      const q = kanjiQuestions[kanjiIndex];
      kanjiInput = [];
      document.getElementById('kanji-qnum').textContent = `${kanjiIndex + 1} / ${kanjiQuestions.length}`;
      document.getElementById('kanji-reading').textContent = `よみ：${q.reading}`;
      document.getElementById('kanji-hint').textContent = q.hint;

      // 解答枠の描画
      const slotsEl = document.getElementById('kanji-slots');
      slotsEl.innerHTML = '';
      for (let i = 0; i < q.word.length; i++) {
        const slot = document.createElement('div');
        slot.className = 'k-slot';
        slot.id = `k-slot-${i}`;
        slotsEl.appendChild(slot);
      }

      // 選択肢タイルの生成（正解の文字＋ダミー文字をシャッフル）
      const chars = shuffle([...q.word.split(''), ...q.dummies]);
      const poolEl = document.getElementById('kanji-pool');
      poolEl.innerHTML = '';
      chars.forEach((ch, idx) => {
        const btn = document.createElement('button');
        btn.className = 'k-tile';
        btn.id = `k-tile-${idx}`;
        btn.textContent = ch;
        btn.onclick = () => selectKanjiTile(ch, idx);
        poolEl.appendChild(btn);
      });
    }

    function selectKanjiTile(ch, tileIdx) {
      const q = kanjiQuestions[kanjiIndex];
      if (kanjiInput.length >= q.word.length) return;

      kanjiInput.push({ char: ch, tileIdx });
      document.getElementById(`k-tile-${tileIdx}`).disabled = true;
      updateKanjiSlots();
    }

    function clearKanjiInput() {
      if (kanjiInput.length === 0) return;
      const last = kanjiInput.pop();
      document.getElementById(`k-tile-${last.tileIdx}`).disabled = false;
      updateKanjiSlots();
    }

    function updateKanjiSlots() {
      const q = kanjiQuestions[kanjiIndex];
      for (let i = 0; i < q.word.length; i++) {
        const slot = document.getElementById(`k-slot-${i}`);
        if (kanjiInput[i]) {
          slot.textContent = kanjiInput[i].char;
          slot.classList.add('filled');
        } else {
          slot.textContent = '';
          slot.classList.remove('filled');
        }
      }
    }

    function submitKanjiAnswer() {
      const q = kanjiQuestions[kanjiIndex];
      const userStr = kanjiInput.map(x => x.char).join('');
      if (userStr.length < q.word.length) {
        showToast('⚠️ まだ空いているマスがあります！', 900);
        return;
      }

      if (userStr === q.word) {
        kanjiScore += 150;
        document.getElementById('kanji-score').textContent = kanjiScore;
        showToast(`⭕ 大正解！「${q.word}」 +150点`, 900);
        kanjiIndex++;
        setTimeout(renderKanjiQuestion, 600);
      } else {
        showToast(`❌ おしい！もう一度よく考えよう`, 900);
        // 入力リセット
        while (kanjiInput.length > 0) clearKanjiInput();
      }
    }

    /* =====================================================================
       モード3：歴史時系列ソート
       ===================================================================== */
    let sortRound = 0;
    const SORT_TOTAL_ROUNDS = 5;
    let sortScore = 0;
    let sortStartTime = 0;
    let currentSortItems = [];
    let draggedIndex = null;

    function startSortGame(eraKey) {
      currentSub = eraKey;
      sortRound = 0;
      sortScore = 0;

      const labels = {
        ancient_medieval: '古代 〜 室町時代',
        modern: '戦国・江戸 〜 現代',
        mix: '全時代ミックス'
      };
      document.getElementById('sort-era').textContent = labels[eraKey];

      showScreen('screen-sort');
      sortStartTime = performance.now();
      timerInterval = setInterval(() => {
        const sec = ((performance.now() - sortStartTime) / 1000).toFixed(1);
        document.getElementById('sort-time').textContent = sec;
      }, 100);

      nextSortRound();
    }

    function nextSortRound() {
      if (sortRound >= SORT_TOTAL_ROUNDS) {
        clearInterval(timerInterval);
        const elapsed = (performance.now() - sortStartTime) / 1000;
        const speedBonus = Math.max(0, Math.round((90 - elapsed) * 4));
        finishGame('歴史時系列ソート', sortScore + speedBonus, `クリアタイム: ${elapsed.toFixed(1)}秒（スピードボーナス +${speedBonus}点）`);
        return;
      }

      sortRound++;
      document.getElementById('sort-qnum').textContent = `${sortRound} / ${SORT_TOTAL_ROUNDS}`;

      let pool = [];
      if (currentSub === 'mix') {
        pool = [...DB.sort.ancient_medieval, ...DB.sort.modern];
      } else {
        pool = [...DB.sort[currentSub]];
      }

      const count = currentDiff === 'normal' ? 4 : 5;
      // すでに正解順になってしまわないようシャッフル
      let picked = shuffle(pool).slice(0, count);
      const isAlreadySorted = picked.every((v, i, a) => i === 0 || a[i - 1].year <= v.year);
      if (isAlreadySorted) picked.reverse();

      currentSortItems = picked;
      renderSortList();
    }

    function renderSortList() {
      const listEl = document.getElementById('sort-list');
      listEl.innerHTML = '';

      currentSortItems.forEach((item, idx) => {
        const row = document.createElement('div');
        row.className = 'sort-item';
        row.draggable = true;

        row.innerHTML = `
          <div class="sort-num">${idx + 1}</div>
          <div class="sort-content">${item.text}</div>
          <div class="sort-arrows">
            <button class="btn-arrow" onclick="moveSortItem(${idx}, -1)" ${idx === 0 ? 'disabled' : ''}>↑</button>
            <button class="btn-arrow" onclick="moveSortItem(${idx}, 1)" ${idx === currentSortItems.length - 1 ? 'disabled' : ''}>↓</button>
          </div>
        `;

        // PC・マウス用ドラッグ＆ドロップイベント
        row.addEventListener('dragstart', () => {
          draggedIndex = idx;
          row.classList.add('dragging');
        });
        row.addEventListener('dragend', () => {
          row.classList.remove('dragging');
          draggedIndex = null;
        });
        row.addEventListener('dragover', (e) => e.preventDefault());
        row.addEventListener('drop', (e) => {
          e.preventDefault();
          if (draggedIndex !== null && draggedIndex !== idx) {
            const moved = currentSortItems.splice(draggedIndex, 1)[0];
            currentSortItems.splice(idx, 0, moved);
            renderSortList();
          }
        });

        listEl.appendChild(row);
      });
    }

    function moveSortItem(index, dir) {
      const target = index + dir;
      if (target < 0 || target >= currentSortItems.length) return;
      const temp = currentSortItems[index];
      currentSortItems[index] = currentSortItems[target];
      currentSortItems[target] = temp;
      renderSortList();
    }

    function checkSortAnswer() {
      let correct = true;
      for (let i = 0; i < currentSortItems.length - 1; i++) {
        if (currentSortItems[i].year > currentSortItems[i + 1].year) {
          correct = false;
          break;
        }
      }

      if (correct) {
        sortScore += 200;
        const yearsSummary = currentSortItems.map(x => `${x.year}年`).join(' → ');
        showToast(`🎉 正解！ +200点<br><span style="font-size:1rem;">(${yearsSummary})</span>`, 1400);
        setTimeout(nextSortRound, 1100);
      } else {
        showToast('❌ 順番が違うところがあるよ！年号を思い出そう', 1200);
      }
    }

    /* =====================================================================
       リザルト画面＆クラス内ローカルランキング（個人情報不使用）
       ===================================================================== */
    function finishGame(modeTitle, finalScore, detailText) {
      document.getElementById('res-title').textContent = `🎉 ${modeTitle} クリア！`;
      document.getElementById('res-score').textContent = finalScore;
      document.getElementById('res-detail').textContent = detailText;

      saveAndRenderRanking(modeTitle, finalScore);
      showScreen('screen-result');
      startConfetti();
    }

    function saveAndRenderRanking(modeTitle, newScore) {
      const storageKey = 'shakai_master_scores_v1';
      let records = [];
      try {
        records = JSON.parse(localStorage.getItem(storageKey)) || [];
      } catch (e) {
        records = [];
      }

      // 個人名は一切保存せず、挑戦番号とモード・スコアのみ記録
      const challengeNo = records.length + 1;
      records.push({
        id: challengeNo,
        mode: modeTitle,
        diff: currentDiff === 'normal' ? '標準' : '上級',
        score: newScore
      });

      // スコア降順で上位5件を保持
      const top5 = [...records].sort((a, b) => b.score - a.score).slice(0, 5);
      try {
        localStorage.setItem(storageKey, JSON.stringify(records));
      } catch (e) {}

      const listEl = document.getElementById('ranking-list');
      listEl.innerHTML = '';
      top5.forEach((rec, idx) => {
        const row = document.createElement('div');
        row.className = 'ranking-row';
        const medal = idx === 0 ? '🥇' : idx === 1 ? '🥈' : idx === 2 ? '🥉' : `${idx + 1}位`;
        row.innerHTML = `
          <span>${medal} 挑戦者 #${rec.id}（${rec.mode}・${rec.diff}）</span>
          <span>${rec.score} 点</span>
        `;
        listEl.appendChild(row);
      });
    }

    function resetRanking() {
      try {
        localStorage.removeItem('shakai_master_scores_v1');
      } catch (e) {}
      document.getElementById('ranking-list').innerHTML = '<div style="padding:8px; color:var(--text-sub);">記録をリセットしました</div>';
    }

    /* =====================================================================
       紙吹雪アニメーション演出
       ===================================================================== */
    let confettiRunning = false;
    let confettiPieces = [];
    const cCanvas = document.getElementById('confetti-canvas');
    const cCtx = cCanvas.getContext('2d');

    function startConfetti() {
      confettiRunning = true;
      cCanvas.width = window.innerWidth;
      cCanvas.height = window.innerHeight;
      confettiPieces = [];
      const palette = ['#005BAB', '#D97706', '#2F855A', '#C53030', '#ECC94B'];
      for (let i = 0; i < 85; i++) {
        confettiPieces.push({
          x: Math.random() * cCanvas.width,
          y: Math.random() * cCanvas.height - cCanvas.height,
          w: Math.random() * 10 + 6,
          h: Math.random() * 6 + 4,
          color: palette[Math.floor(Math.random() * palette.length)],
          vy: Math.random() * 3 + 2,
          vx: Math.random() * 2 - 1,
          rot: Math.random() * 360,
          vrot: Math.random() * 6 - 3
        });
      }
      requestAnimationFrame(loopConfetti);
    }

    function stopConfetti() {
      confettiRunning = false;
      cCtx.clearRect(0, 0, cCanvas.width, cCanvas.height);
    }

    function loopConfetti() {
      if (!confettiRunning) return;
      cCtx.clearRect(0, 0, cCanvas.width, cCanvas.height);
      confettiPieces.forEach(p => {
        p.y += p.vy;
        p.x += p.vx;
        p.rot += p.vrot;
        if (p.y > cCanvas.height) {
          p.y = -15;
          p.x = Math.random() * cCanvas.width;
        }
        cCtx.save();
        cCtx.translate(p.x, p.y);
        cCtx.rotate((p.rot * Math.PI) / 180);
        cCtx.fillStyle = p.color;
        cCtx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
        cCtx.restore();
      });
      requestAnimationFrame(loopConfetti);
    }

    window.addEventListener('resize', () => {
      if (confettiRunning) {
        cCanvas.width = window.innerWidth;
        cCanvas.height = window.innerHeight;
      }
    });
  </script>
