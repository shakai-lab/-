// 文字列からハッシュ生成（クラスコードによる擬似乱数シード）
function createSeededRandom(seedStr) {
  if (!seedStr) return Math.random;
  let h = 2166136261 >>> 0;
  for (let i = 0; i < seedStr.length; i++) {
    h ^= seedStr.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return function() {
    h += 0x6D2B79F5;
    let t = Math.imul(h ^ (h >>> 15), 1 | h);
    t ^= t + Math.imul(t ^ (t >>> 7), 61 | t);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// 配列シャッフル（シード対応）
function shuffleArray(array, rngFunc) {
  const arr = [...array];
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(rngFunc() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

function getFilteredPool() {
  if (config.era === "all") return [...HISTORY_DATA];
  return HISTORY_DATA.filter(item => item.era === config.era);
}

function startGame() {
  stopTimer();
  config.nickname = document.getElementById("input-nickname").value.trim() || "歴史チャレンジャー";
  config.roomCode = document.getElementById("input-roomcode").value.trim();

  // シード値の初期化（コード + モード + 時代 + 時間）
  const seedKey = config.roomCode ? `${config.roomCode}_${config.mode}_${config.era}_${config.timeLimit}` : "";
  gameState = {
    score: 0,
    combo: 0,
    maxCombo: 0,
    timeLeft: config.timeLimit,
    timerId: null,
    isPlaying: true,
    rng: createSeededRandom(seedKey),
    stage: 1,
    flippedCards: [],
    matchedPairs: 0,
    totalPairs: 0,
    isLockBoard: false,
    totalClearedPairs: 0,
    qNum: 1,
    correctQuestions: 0,
    currentSortItems: [],
    isCheckingSort: false
  };

  if (config.mode === "memory") {
    showScreen("screen-memory");
    updateMemoryHUD();
    initMemoryStage();
  } else {
    showScreen("screen-sort");
    updateSortHUD();
    initSortQuestion();
  }

  startTimer();
}

function startTimer() {
  updateTimerDisplay();
  gameState.timerId = setInterval(() => {
    gameState.timeLeft--;
    updateTimerDisplay();
    if (gameState.timeLeft <= 0) {
      endGame();
    }
  }, 1000);
}

function stopTimer() {
  if (gameState.timerId) {
    clearInterval(gameState.timerId);
    gameState.timerId = null;
  }
}

function updateTimerDisplay() {
  const m = String(Math.floor(gameState.timeLeft / 60)).padStart(2, "0");
  const s = String(gameState.timeLeft % 60).padStart(2, "0");
  const text = `${m}:${s}`;
  document.getElementById("mem-timer").textContent = text;
  document.getElementById("sort-timer").textContent = text;
}

  el.style.border = `4px solid ${color}`;
  el.classList.add("show");

}

function updateMemoryHUD() {
  document.getElementById("mem-stage").textContent = gameState.stage;
  document.getElementById("mem-score").textContent = gameState.score;
  const comboEl = document.getElementById("mem-combo");
  if (gameState.combo >= 2) {
    comboEl.textContent = `${gameState.combo}連続ボーナス!`;
    comboEl.style.display = "inline-block";
  } else {
    comboEl.style.display = "none";
  }
}

function initMemoryStage() {
  const board = document.getElementById("memory-board");
  board.innerHTML = "";
  gameState.flippedCards = [];
  gameState.matchedPairs = 0;
  gameState.isLockBoard = false;

  // ステージが進むごとにペア数を増やす（4ペア → 6ペア → 最大8ペア）
  let pairCount = 4;
  if (gameState.stage === 2) pairCount = 6;
  else if (gameState.stage >= 3) pairCount = 8;

  const pool = shuffleArray(getFilteredPool(), gameState.rng);
  const selectedItems = pool.slice(0, Math.min(pairCount, pool.length));
  gameState.totalPairs = selectedItems.length;

  // カードデータ生成（年号カードと出来事カード）
  let cards = [];
  selectedItems.forEach((item, idx) => {
    cards.push({ id: idx, type: "year", text: item.yearText });
    cards.push({ id: idx, type: "event", text: item.event });
  });

  cards = shuffleArray(cards, gameState.rng);

  cards.forEach(cardData => {
    const cardEl = document.createElement("div");
    cardEl.className = `memory-card hidden-card type-${cardData.type}`;
    cardEl.dataset.id = cardData.id;
    cardEl.dataset.type = cardData.type;
    cardEl.dataset.text = cardData.text;

    cardEl.addEventListener("click", () => handleCardClick(cardEl));
    board.appendChild(cardEl);
  });
}

function handleCardClick(cardEl) {
  if (!gameState.isPlaying || gameState.isLockBoard) return;
  if (cardEl.classList.contains("flipped") || cardEl.classList.contains("matched")) return;

  // カードを表にする
  cardEl.classList.remove("hidden-card");
  cardEl.classList.add("flipped");
  cardEl.textContent = cardEl.dataset.text;
  gameState.flippedCards.push(cardEl);

  if (gameState.flippedCards.length === 2) {
    checkMemoryMatch();
  }
}

function checkMemoryMatch() {
  gameState.isLockBoard = true;
  const [card1, card2] = gameState.flippedCards;

  // 同じIDかつ異なるタイプ（年号と出来事）なら正解
  const isMatch = (card1.dataset.id === card2.dataset.id) && (card1.dataset.type !== card2.dataset.type);

  if (isMatch) {
    gameState.combo++;
    if (gameState.combo > gameState.maxCombo) gameState.maxCombo = gameState.combo;

    const bonus = (gameState.combo - 1) * 50;
    const earned = 100 + bonus;
    gameState.score += earned;
    gameState.matchedPairs++;
    gameState.totalClearedPairs++;
    updateMemoryHUD();

    showFeedback(`⭕ 正解！ +${earned}pt`, "#00796B");

    setTimeout(() => {
      card1.classList.add("matched");
      card2.classList.add("matched");
      gameState.flippedCards = [];
      gameState.isLockBoard = false;

      // ステージクリア判定
      if (gameState.matchedPairs >= gameState.totalPairs && gameState.isPlaying) {
        const stageBonus = gameState.stage * 200;
        gameState.score += stageBonus;
        gameState.stage++;
        updateMemoryHUD();
        launchConfetti();
        showFeedback(`🎉 ステージクリア！ +${stageBonus}pt`, "#E65100");
        setTimeout(() => {
          if (gameState.isPlaying) initMemoryStage();
        }, 900);
      }
    }, 550);
  } else {
    // 不正解
    gameState.combo = 0;
    updateMemoryHUD();
    setTimeout(() => {
      card1.classList.remove("flipped");
      card1.classList.add("hidden-card");
      card1.textContent = "";
      card2.classList.remove("flipped");
      card2.classList.add("hidden-card");
      card2.textContent = "";
      gameState.flippedCards = [];
      gameState.isLockBoard = false;
    }, 850);
  }
}

function initSortQuestion() {
  gameState.isCheckingSort = false;
  const btn = document.getElementById("btn-check-sort");
  btn.disabled = false;
  btn.textContent = "✅ この順番で決定！";

  // 重複しない年号の出来事を5つ抽出
  const pool = shuffleArray(getFilteredPool(), gameState.rng);
  const uniqueYearItems = [];
  const usedYears = new Set();

  for (const item of pool) {
    if (!usedYears.has(item.year)) {
      usedYears.add(item.year);
      uniqueYearItems.push(item);
      if (uniqueYearItems.length === 5) break;
    }
  }

  // 最初から正解順になっていないようシャッフル
  let shuffled = shuffleArray(uniqueYearItems, gameState.rng);
  const sortedYears = [...uniqueYearItems].sort((a, b) => a.year - b.year).map(i => i.year).join(",");
  if (shuffled.map(i => i.year).join(",") === sortedYears) {
    shuffled.reverse();
  }

  gameState.currentSortItems = shuffled;
  renderSortList();
}

function renderSortList() {
  const listEl = document.getElementById("sort-list");
  listEl.innerHTML = "";

  gameState.currentSortItems.forEach((item, index) => {
    const row = document.createElement("div");
    row.className = "sort-item";
    row.dataset.year = item.year;

    row.innerHTML = `
      <div class="drag-handle">☰</div>
      <div class="sort-rank">${index + 1}</div>
      <div class="sort-text">${item.event}</div>
      <div class="sort-year-reveal">${item.yearText}</div>
    `;

    setupTouchDrag(row, listEl);
    listEl.appendChild(row);
  });
}

// タブレット・マウス両対応のPointer Eventsによる直感的な並べ替え
function setupTouchDrag(itemEl, containerEl) {
  itemEl.addEventListener("pointerdown", (e) => {
    if (gameState.isCheckingSort || !gameState.isPlaying) return;
    e.preventDefault();
    itemEl.setPointerCapture(e.pointerId);
    itemEl.classList.add("dragging");

    const onPointerMove = (moveEvent) => {
      const siblings = [...containerEl.querySelectorAll(".sort-item:not(.dragging)")];
      const nextSibling = siblings.find(sib => {
        const rect = sib.getBoundingClientRect();
        return moveEvent.clientY < rect.top + rect.height / 2;
      });

      if (nextSibling) {
        containerEl.insertBefore(itemEl, nextSibling);
      } else {
        containerEl.appendChild(itemEl);
      }
      updateSortRanks();
    };

    const onPointerUp = () => {
      itemEl.classList.remove("dragging");
      itemEl.removeEventListener("pointermove", onPointerMove);
      itemEl.removeEventListener("pointerup", onPointerUp);
      itemEl.removeEventListener("pointercancel", onPointerUp);
      updateSortRanks();
    };

    itemEl.addEventListener("pointermove", onPointerMove);
    itemEl.addEventListener("pointerup", onPointerUp);
    itemEl.addEventListener("pointercancel", onPointerUp);
  });
}

function updateSortRanks() {
  const rows = document.querySelectorAll("#sort-list .sort-item");
  rows.forEach((row, idx) => {
    row.querySelector(".sort-rank").textContent = idx + 1;
  });
}

function checkSortAnswer() {
  if (gameState.isCheckingSort || !gameState.isPlaying) return;
  gameState.isCheckingSort = true;

  const btn = document.getElementById("btn-check-sort");
  btn.disabled = true;

  const rows = [...document.querySelectorAll("#sort-list .sort-item")];
  const userYears = rows.map(r => parseInt(r.dataset.year, 10));
  const correctYears = [...userYears].sort((a, b) => a - b);

  let correctCount = 0;
  rows.forEach((row, idx) => {
    const revealEl = row.querySelector(".sort-year-reveal");
    revealEl.style.display = "block";
    if (userYears[idx] === correctYears[idx]) {
      row.classList.add("correct-pos");
      correctCount++;
    } else {
      row.classList.add("wrong-pos");
    }
  });

  if (correctCount === 5) {
    gameState.combo++;
    if (gameState.combo > gameState.maxCombo) gameState.maxCombo = gameState.combo;
    gameState.correctQuestions++;

    const bonus = (gameState.combo - 1) * 100;
    const earned = 300 + bonus;
    gameState.score += earned;
    launchConfetti();
    showFeedback(`🎉 パーフェクト！ +${earned}pt`, "#00796B");
  } else {
    gameState.combo = 0;
    const partial = correctCount * 30;
    gameState.score += partial;
    showFeedback(`惜しい！${correctCount}個一致 (+${partial}pt)`, "#E65100");
  }

  updateSortHUD();

  setTimeout(() => {
    if (gameState.isPlaying) {
      gameState.qNum++;
      updateSortHUD();
      initSortQuestion();
    }
  }, 1800);
}

}

// 軽量紙吹雪アニメーション
function launchConfetti() {
  const canvas = document.getElementById("confetti-canvas");
  const ctx = canvas.getContext("2d");
  canvas.width = window.innerWidth;
  canvas.height = window.innerHeight;

  const colors = ["#005BAB", "#E65100", "#00796B", "#FBC02D", "#D32F2F"];
  const particles = Array.from({ length: 65 }, () => ({
    x: Math.random() * canvas.width,
    y: Math.random() * canvas.height - canvas.height,
    r: Math.random() * 6 + 4,
    d: Math.random() * 5 + 2,
    color: colors[Math.floor(Math.random() * colors.length)],
    tilt: Math.random() * 10 - 5
  }));

  let frame = 0;
  function draw() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    particles.forEach(p => {
      ctx.fillStyle = p.color;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
      ctx.fill();
      p.y += p.d;
      p.x += Math.sin(frame * 0.05) * 1.5;
    });
    frame++;
    if (frame < 75) {
      requestAnimationFrame(draw);
    } else {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
    }
  }
  draw();
}
